function sessionForUser(userId) {
  return db.prepare(`
    SELECT u.id, u.name, u.email, u.avatar_url,
           c.id AS couple_id, c.name AS couple_name, c.relationship_start, c.invite_code, c.pin_hash,
           cm.role, cm.joined_at
    FROM users u
    JOIN couple_members cm ON cm.user_id=u.id
    JOIN couples c ON c.id=cm.couple_id
    WHERE u.id=?
  `).get(userId);
}

function requireAuth(req, res, next) {
  try {
    const token = req.cookies.zawsze_session;
    if (!token) return res.status(401).json({ error: 'Please sign in.' });
    const payload = jwt.verify(token, JWT_SECRET);
    if (payload.scope !== 'session') throw new Error('Invalid scope');
    const session = sessionForUser(Number(payload.sub));
    if (!session) return res.status(401).json({ error: 'Session no longer exists.' });
    req.auth = session;
    next();
  } catch {
    return res.status(401).json({ error: 'Your session has expired. Please sign in again.' });
  }
}

function isUnlocked(req) {
  if (!req.auth?.pin_hash) return true;
  const token = req.cookies.zawsze_unlock;
  if (!token) return false;
  try {
    const p = jwt.verify(token, JWT_SECRET);
    return p.scope === 'unlock' && Number(p.coupleId) === Number(req.auth.couple_id);
  } catch { return false; }
}

function requireOwner(req, res, next) {
  if (req.auth.role !== 'owner') return res.status(403).json({ error: 'Only the space owner can do this.' });
  next();
}

function notifyPartner(coupleId, actorId, type, entityType, entityId, message) {
  const partners = db.prepare('SELECT user_id FROM couple_members WHERE couple_id=? AND user_id<>?').all(coupleId, actorId);
  const stmt = db.prepare(`INSERT INTO notifications(couple_id,target_user_id,actor_id,type,entity_type,entity_id,message) VALUES(?,?,?,?,?,?,?)`);
  for (const p of partners) stmt.run(coupleId, p.user_id, actorId, type, entityType || null, entityId || null, message);
}

function fileUrl(filePath) {
  return `/api/media/${encodeURIComponent(path.basename(filePath))}`;

function hydrateFiles(rows, idKey) {
  const ids = rows.map(r => r.id);
  if (!ids.length) return rows.map(r => ({ ...r, files: [] }));
  const placeholders = ids.map(() => '?').join(',');
  const table = idKey === 'memory_id' ? 'memory_files' : 'timeline_files';
  const files = db.prepare(`SELECT id, ${idKey} AS parent_id, file_path, original_name, mime_type, position FROM ${table} WHERE ${idKey} IN (${placeholders}) ORDER BY position,id`).all(...ids);
  const by = new Map();
  for (const f of files) {
    const item = { id: f.id, url: fileUrl(f.file_path), originalName: f.original_name, mimeType: f.mime_type, position: f.position };
    if (!by.has(f.parent_id)) by.set(f.parent_id, []);
    by.get(f.parent_id).push(item);
  }
  return rows.map(r => ({ ...r, files: by.get(r.id) || [] }));
}

const storage = multer.diskStorage({
  destination(req, file, cb) {
    const dir = path.join(uploadRoot, String(req.auth.couple_id));
    fs.mkdirSync(dir, { recursive: true });
    cb(null, dir);
  },
  filename(req, file, cb) {
    const ext = path.extname(file.originalname).toLowerCase().replace(/[^.a-z0-9]/g, '').slice(0, 8);
    cb(null, `${crypto.randomUUID()}${ext || '.img'}`);
  }
});
const upload = multer({
  storage,
  limits: { fileSize: MAX_UPLOAD_MB * 1024 * 1024, files: 12 },
  fileFilter(req, file, cb) {
    if (!/^image\/(jpeg|png|webp|gif|avif)$/i.test(file.mimetype)) return cb(new Error('Only JPEG, PNG, WebP, GIF, or AVIF images are allowed.'));
    cb(null, true);
  }
});
