    db.prepare("UPDATE couples SET pin_hash=NULL,updated_at=datetime('now') WHERE id=?").run(req.auth.couple_id);
    res.clearCookie('zawsze_unlock', { httpOnly:true,sameSite:'strict',secure:COOKIE_SECURE });
    return res.json({ ok: true, hasPin: false });
  }
  if (!/^\d{4,8}$/.test(pin)) return res.status(400).json({ error: 'Use a 4–8 digit PIN.' });
  const hash = await bcrypt.hash(pin, 12);
  db.prepare("UPDATE couples SET pin_hash=?,updated_at=datetime('now') WHERE id=?").run(hash, req.auth.couple_id);
  res.json({ ok: true, hasPin: true });
});

  const token = jwt.sign({ scope: 'unlock', coupleId: req.auth.couple_id }, JWT_SECRET, { expiresIn: '15m' });
  res.cookie('zawsze_unlock', token, { httpOnly:true,sameSite:'strict',secure:COOKIE_SECURE,maxAge:15*60*1000 });
  res.json({ ok: true, expiresInMinutes: 15 });
});
